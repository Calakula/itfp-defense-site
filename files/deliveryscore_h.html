#ifndef DELIVERYSCORE_H
#define DELIVERYSCORE_H

#include <stdint.h>

#ifdef __cplusplus
extern "C" {
#endif

/* Export macro: Windows needs dllexport, other platforms need default visibility. */
#if defined(_WIN32)
#define DQ_API __declspec(dllexport)
#else
#define DQ_API __attribute__((visibility("default")))
#endif

/* Valid input range of dq_load_points (inclusive). */
#define DQ_MAX_DISTANCE_M 100000  /* 100 km in metres */
#define DQ_MAX_WEIGHT_G   100000  /* 100 kg in grams  */

/*
 * Courier effort points of one delivery.
 *   distance_m : distance in whole metres, 0 .. DQ_MAX_DISTANCE_M
 *   weight_g   : parcel weight in whole grams, 0 .. DQ_MAX_WEIGHT_G
 * Returns distance_m / 100 + weight_g / 250 (integer division, rounded down):
 * one point for every full 100 m and one point for every full 250 g.
 * Returns -1 if an argument is outside its range. No memory is allocated.
 * Example: dq_load_points(2000, 1000) = 20 + 4 = 24.
 */
DQ_API int32_t dq_load_points(int32_t distance_m, int32_t weight_g);

#ifdef __cplusplus
}
#endif

#endif /* DELIVERYSCORE_H */
